      // ===== QUOTES (free API via exchangerate / fallback random walk) =====
      const lastPrice = {};
      function seedPrice(sym){
      const seeds = {'FX:EURUSD':1.0874,'FX:GBPUSD':1.2710,'FX:USDJPY':156.42,'OANDA:XAUUSD':2340.10,'TVC:USOIL':78.95,'SP:SPX':5231.40,'NASDAQ:AAPL':196.45,'NASDAQ:TSLA':242.10,'NASDAQ:NVDA':1180.5,'BITSTAMP:BTCUSD':67420,'BITSTAMP:ETHUSD':3512};
      return seeds[sym] || 100;
      }
      function tick(sym){
      const p = lastPrice[sym] ?? seedPrice(sym);
      const vol = p * 0.0006;
      const next = +(p + (Math.random()-0.5)*vol).toFixed(p>100?2:5);
      lastPrice[sym] = next;
      return next;
      }
      function refreshQuote(){
      const px = tick(currentSymbol);
      const spread = px * 0.0001;
      document.getElementById('bid').textContent = (px-spread).toFixed(px>100?2:5);
      document.getElementById('ask').textContent = (px+spread).toFixed(px>100?2:5);
      }

      return false;

      function placeOrder(side){
      const u=currentUser();
      if(!u){ openAuth('demo'); toast('Sign in or open a demo first',true); return; }
      const vol=parseFloat(document.getElementById('vol').value);
      if(!vol||vol<=0){ toast('Enter a valid volume',true); return; }
      const px = lastPrice[currentSymbol] ?? seedPrice(currentSymbol);
      const positions=getJSON(KEYS.positions,{});
      const list=positions[u.email]||[];
      list.push({
      id: Date.now(), symbol: currentSymbol, side, vol,
      open: px, sl: parseFloat(document.getElementById('sl').value)||null,
      tp: parseFloat(document.getElementById('tp').value)||null,
      ts: Date.now()
      });
      positions[u.email]=list; setJSON(KEYS.positions,positions);
      toast(`${side} ${vol} ${currentSymbol} @ ${px}`);
      updatePositions();
      }
      function contractSize(sym){ if(sym.startsWith('FX:')||sym.startsWith('OANDA:')) return 100000; if(sym.includes('BTC')||sym.includes('ETH')) return 1; return 100; }
      function pnl(pos){
      const px = lastPrice[pos.symbol] ?? seedPrice(pos.symbol);
      const diff = (px - pos.open) * (pos.side==='BUY'?1:-1);
      return diff * pos.vol * contractSize(pos.symbol) * (pos.symbol.includes('JPY')?0.0067:1) / (pos.symbol.startsWith('FX:')?1:1);
      }
      function closePosition(id){
      const u=currentUser(); if(!u) return;
      const positions=getJSON(KEYS.positions,{});
      const list=positions[u.email]||[]; const p=list.find(x=>x.id===id); if(!p) return;
      const profit=pnl(p);
      const users=getJSON(KEYS.users,{}); users[u.email].balance += profit; setJSON(KEYS.users,users);
      positions[u.email]=list.filter(x=>x.id!==id); setJSON(KEYS.positions,positions);
      toast(`Closed ${p.symbol} ${profit>=0?'+':''}${profit.toFixed(2)}`);
      renderSession();
      }
      function updatePositions(){
      const u=currentUser();
      const tbody=document.getElementById('positionsBody');
      if(!u){ tbody.innerHTML=`<tr><td colspan="7" class="py-6 text-center text-gray-500">Sign in to view positions</td></tr>`; document.getElementById('pnlSummary').innerHTML='P/L: <span class="text-white">$0.00</span>'; return; }
      const list=(getJSON(KEYS.positions,{})[u.email])||[];
      if(!list.length){ tbody.innerHTML=`<tr><td colspan="7" class="py-6 text-center text-gray-500">No open positions</td></tr>`; document.getElementById('pnlSummary').innerHTML='P/L: <span class="text-white">$0.00</span>'; return; }
      let total=0;
      tbody.innerHTML=list.map(p=>{
      const cur=lastPrice[p.symbol]??seedPrice(p.symbol); const pl=pnl(p); total+=pl;
      const col=pl>=0?'text-brand-green':'text-red-400';
      return `<tr class="border-b border-line/50">
      <td class="py-2">${p.symbol}</td>
      <td class="${p.side==='BUY'?'text-brand-green':'text-red-400'}">${p.side}</td>
      <td class="text-right">${p.vol}</td>
      <td class="text-right">${p.open}</td>
      <td class="text-right">${cur}</td>
      <td class="text-right ${col}">${pl>=0?'+':''}${pl.toFixed(2)}</td>
      <td class="text-right"><button onclick="closePosition(${p.id})" class="text-xs px-2 py-1 rounded bg-white/5 hover:bg-white/10">Close</button></td>
      </tr>`;
      }).join('');
      const col=total>=0?'text-brand-green':'text-red-400';
      document.getElementById('pnlSummary').innerHTML=`P/L: <span class="${col}">${total>=0?'+':''}$${total.toFixed(2)}</span>`;
      }
